async function* getRemoteAgentOverviewsStream(url, signal) {
    let response;
    let reader;

    try {
        /*************************************************************************
         * Fetch API call with abort signal
         * If request aborts, will throw AbortError
         *************************************************************************/
        response = await fetch(url, { signal });

        /*************************************************************************
         * Acquire ReadableStream reader
         * Needed to manually process streaming JSON lines
         *************************************************************************/
        reader = response.body.getReader();

        let decoder = new TextDecoder('utf-8');
        let buffer = '';

        while (true) {
            /*************************************************************************
             * Read next chunk from the stream
             * Each chunk is Uint8Array
             *************************************************************************/
            const { value, done } = await reader.read();

            if (done) break;

            buffer += decoder.decode(value, { stream: true });

            // Split by newlines, yield one JSON per line
            let lines = buffer.split('\n');
            buffer = lines.pop(); // remainder

            for (const line of lines) {
                if (!line.trim()) continue;
                try {
                    yield JSON.parse(line);
                } catch (err) {
                    console.error('JSON parse error', err, line);
                }
            }
        }

        // Yield final leftover chunk if any
        if (buffer.trim()) {
            try {
                yield JSON.parse(buffer);
            } catch (err) {
                console.error('JSON parse error', err, buffer);
            }
        }

    } catch (err) {
        /*************************************************************************
         * Catch network or AbortError
         * This is where retry logic could be implemented
         *************************************************************************/
        if (err.name === 'AbortError') {
            console.warn('Fetch request aborted:', err.message);
        } else {
            console.error('Fetch failed:', err);
        }
        throw err; // rethrow so calling code knows

    } finally {
        /*************************************************************************
         * CLEANUP BLOCK
         * Ensures ReadableStream reader is released
         * Prevents memory leaks and runaway zygote CPU usage
         *************************************************************************/
        if (reader) {
            try {
                await reader.cancel();
            } catch (cancelErr) {
                console.warn('Reader cancel failed:', cancelErr);
            }
        }

        if (response && response.body) {
            try {
                response.body.cancel();
            } catch (bodyErr) {
                console.warn('Response body cancel failed:', bodyErr);
            }
        }
    }
}

#!/usr/bin/env node
/***************************************************************************************************
TITLE:
PRF-COMPLIANT FETCH STREAM WITH RETRY AND ABORT CLEANUP

PURPOSE:
- Retry failed streaming fetches automatically
- Handle AbortError gracefully
- Ensure ReadableStream readers are released each attempt
- Fully PRF-compliant with inline verbose comments

- Node.js >=18 required

/***************************************************************************************************
ASYNC GENERATOR FUNCTION: getRemoteAgentOverviewsStreamWithRetry
- Wraps the basic streaming fetch with retry logic
- Ensures cleanup between attempts to prevent leaks
***************************************************************************************************/
async function* getRemoteAgentOverviewsStreamWithRetry(url, signal, options = {}) {
    const maxRetries = options.maxRetries ?? 3;
    const backoffMs = options.backoffMs ?? 1000;

    let attempt = 0;

    while (attempt <= maxRetries) {
        let response;
        let reader;

        try {
            /*************************************************************************
             * Increment attempt counter
             *************************************************************************/
            attempt++;

            /*************************************************************************
             * Fetch API call with abort signal
             *************************************************************************/
            response = await fetch(url, { signal });

            /*************************************************************************
             * If response not OK, throw to trigger retry
             *************************************************************************/
            if (!response.ok) {
                throw new Error(`HTTP error ${response.status}`);
            }

            /*************************************************************************
             * Acquire ReadableStream reader
             *************************************************************************/
            reader = response.body.getReader();
            const decoder = new TextDecoder('utf-8');
            let buffer = '';

            /*************************************************************************
             * Stream reading loop
             *************************************************************************/
            while (true) {
                const { value, done } = await reader.read();
                if (done) break;
                buffer += decoder.decode(value, { stream: true });

                let lines = buffer.split('\n');
                buffer = lines.pop();

                for (const line of lines) {
                    if (!line.trim()) continue;
                    try {
                        yield JSON.parse(line);
                    } catch (err) {
                        console.error('JSON parse error', err, line);
                    }
                }
            }

            if (buffer.trim()) {
                try {
                    yield JSON.parse(buffer);
                } catch (err) {
                    console.error('JSON parse error', err, buffer);
                }
            }

            /*************************************************************************
             * Success exit, no retry needed
             *************************************************************************/
            break;

        } catch (err) {
            /*************************************************************************
             * Handle network or AbortError
             *************************************************************************/
            if (err.name === 'AbortError') {
                console.warn('Fetch aborted:', err.message);
                throw err; // abort is terminal
            } else {
                console.warn(`Attempt ${attempt} failed:`, err.message);
                if (attempt > maxRetries) throw err;

                /*************************************************************************
                 * Optional backoff before retry
                 *************************************************************************/
                await new Promise(r => setTimeout(r, backoffMs * attempt));
            }

        } finally {
            /*************************************************************************
             * Cleanup to prevent leaking readers / response body
             *************************************************************************/
            if (reader) {
                try { await reader.cancel(); } catch (cancelErr) { console.warn('Reader cancel failed', cancelErr); }
            }
            if (response && response.body) {
                try { response.body.cancel(); } catch (bodyErr) { console.warn('Response body cancel failed', bodyErr); }
            }
        }
    }
}
