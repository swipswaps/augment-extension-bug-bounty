#!/usr/bin/env node

const fs = require("fs");
const path = require("path");
const os = require("os");

function findAugmentExtension() {
    const base = path.join(os.homedir(), ".vscode", "extensions");
    const entries = fs.readdirSync(base);
    const match = entries.find(e => e.startsWith("augment.vscode-augment"));
    if (!match) throw new Error("Augment extension not found.");
    return path.join(base, match, "out", "extension.js");
}

function patchFile(filePath) {
    const original = fs.readFileSync(filePath, "utf8");

    if (!original.includes("async cancelToolRun"))
        throw new Error("cancelToolRun not found.");

    if (!original.includes("await this.close(!0)"))
        throw new Error("Expected close(!0) call not found.");

    const backupPath = filePath + ".backup-" + Date.now();
    fs.writeFileSync(backupPath, original);
    console.log("Backup created:", backupPath);

    const patched = original.replace(
        /async cancelToolRun\s*\([^\)]*\)\s*\{[^}]*?await this\.close\(!0\)[^}]*?\}/s,
`async cancelToolRun(t, r){
    if (!this.isRequestActive(t, r)) return false;

    // SAFE PATCH: Do NOT close MCP host for tool cancellation
    this._cancelledByUser = true;

    if (this._runningTool && this._runningTool.abortController) {
        try { this._runningTool.abortController.abort(); } catch {}
    }

    this._runningTool = void 0;
    return true;
}`
    );

    fs.writeFileSync(filePath, patched);
    console.log("Patch applied successfully.");
}

try {
    const extPath = findAugmentExtension();
    console.log("Found extension:", extPath);
    patchFile(extPath);
    console.log("\nReload VS Code window now.");
} catch (err) {
    console.error("Patch failed:", err.message);
    process.exit(1);
}
