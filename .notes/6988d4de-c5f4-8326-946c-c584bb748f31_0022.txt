#!/usr/bin/env node

const fs = require('fs');
const path = require('path');
const { spawn } = require('child_process');

const LOG_ROOT = path.join(
  process.env.HOME,
  '.config',
  'Code',
  'logs'
);

const TIMEOUT_REGEX = /timeout|cancelled due to timeout/i;
const EXT_HOST_REGEX = /extension host/i;
const UNHANDLED_REGEX = /unhandled|rejection|exception/i;

let lastEventLoopTick = Date.now();
let lastExtensionHostPid = null;

function getLatestSessionDir() {
  const dirs = fs.readdirSync(LOG_ROOT)
    .map(d => path.join(LOG_ROOT, d))
    .filter(p => fs.statSync(p).isDirectory())
    .sort((a, b) => fs.statSync(b).mtimeMs - fs.statSync(a).mtimeMs);

  return dirs[0];
}

function structuredReport(type, details) {
  const report = {
    timestamp: new Date().toISOString(),
    type,
    details
  };

  console.log(JSON.stringify(report, null, 2));
}

function monitorLogs() {
  const sessionDir = getLatestSessionDir();
  const exthostLog = path.join(sessionDir, 'exthost.log');
  const mainLog = path.join(sessionDir, 'main.log');

  console.log(`[INFO] Monitoring logs in: ${sessionDir}`);

  [exthostLog, mainLog].forEach(file => {
    if (!fs.existsSync(file)) return;

    const tail = spawn('tail', ['-F', file]);

    tail.stdout.on('data', data => {
      const lines = data.toString().split('\n');

      lines.forEach(line => {
        if (TIMEOUT_REGEX.test(line)) {
          structuredReport('TIMEOUT_DETECTED', { line });
        }

        if (UNHANDLED_REGEX.test(line)) {
          structuredReport('UNHANDLED_ERROR', { line });
        }

        if (EXT_HOST_REGEX.test(line) && /restart|crash|terminated/i.test(line)) {
          structuredReport('EXTENSION_HOST_EVENT', { line });
        }
      });
    });
  });
}

function monitorExtensionHostPid() {
  setInterval(() => {
    const ps = spawn('pgrep', ['-f', 'extensionHost']);

    let output = '';

    ps.stdout.on('data', data => {
      output += data.toString();
    });

    ps.on('close', () => {
      const pids = output.trim().split('\n').filter(Boolean);

      if (pids.length === 0) {
        structuredReport('EXTENSION_HOST_NOT_RUNNING', {});
        return;
      }

      if (!lastExtensionHostPid) {
        lastExtensionHostPid = pids[0];
      }

      if (pids[0] !== lastExtensionHostPid) {
        structuredReport('EXTENSION_HOST_RESTARTED', {
          oldPid: lastExtensionHostPid,
          newPid: pids[0]
        });
        lastExtensionHostPid = pids[0];
      }
    });

  }, 2000);
}

function monitorEventLoop() {
  const interval = 100;
  setInterval(() => {
    const now = Date.now();
    const drift = now - lastEventLoopTick - interval;

    if (drift > 250) {
      structuredReport('EVENT_LOOP_STALL', {
        driftMs: drift
      });
    }

    lastEventLoopTick = now;
  }, interval);
}

function main() {
  if (!fs.existsSync(LOG_ROOT)) {
    console.error('[ERROR] VS Code log directory not found.');
    process.exit(1);
  }

  monitorLogs();
  monitorExtensionHostPid();
  monitorEventLoop();
}

main();

Session directory changes

Continue tailing old files

Failure Mode B — Tail Process Dies

Using tail -F via spawn():

If tail crashes

If file descriptor breaks

If OS hits resource limits

Your monitor silently loses visibility.

Failure Mode C — Event Loop Monitoring Self-Contamination

Your script measures event loop stall in itself.

Not in VS Code’s extension host.

So:

It detects stalls in the monitor process

Not necessarily stalls in the extension host

False confidence.

Failure Mode D — PID Race Condition

Using pgrep -f extensionHost:

Multiple extension hosts may exist

PIDs can recycle

Short-lived restarts may be missed

This produces incorrect restart detection.

Failure Mode E — Regex False Positives

This pattern:

/timeout|cancelled due to timeout/i


Matches:

Debug logs

Non-critical timeouts

Unrelated text

You may correlate noise as root cause.

Failure Mode F — Blocking I/O

Heavy logging + JSON stringify:

Can block your monitor event loop

Cause self-induced stall

Distort timing

Your script may consume memory

Stream buffers grow

#!/usr/bin/env node

'use strict';

const fs = require('fs');
const path = require('path');
const { spawn } = require('child_process');
const readline = require('readline');

const LOG_ROOT = path.join(process.env.HOME, '.config', 'Code', 'logs');

const STRICT_TIMEOUT_REGEX =
  /\bTool call was cancelled due to timeout\b/;

const STRICT_UNHANDLED_REGEX =
  /\b(unhandled rejection|uncaught exception)\b/i;

let currentSessionDir = null;
let extensionHostPid = null;

function now() {
  return Number(process.hrtime.bigint() / 1000000n); // monotonic ms
}

function report(type, details) {
  const payload = {
    monotonicMs: now(),
    isoTime: new Date().toISOString(),
    type,
    details
  };
  process.stdout.write(JSON.stringify(payload) + '\n');
}

function getLatestSessionDir() {
  const dirs = fs.readdirSync(LOG_ROOT)
    .map(d => path.join(LOG_ROOT, d))
    .filter(d => fs.statSync(d).isDirectory())
    .sort((a, b) => fs.statSync(b).mtimeMs - fs.statSync(a).mtimeMs);

  return dirs[0] || null;
}

function watchLogFile(filePath) {
  let position = 0;

  function readNew() {
    fs.stat(filePath, (err, stats) => {
      if (err) return;

      if (stats.size < position) {
        // log rotated
        position = 0;
      }

      const stream = fs.createReadStream(filePath, {
        start: position,
        end: stats.size
      });

      const rl = readline.createInterface({
        input: stream
      });

      rl.on('line', line => {
        if (STRICT_TIMEOUT_REGEX.test(line)) {
          report('TIMEOUT_DETECTED', { line });
        }

        if (STRICT_UNHANDLED_REGEX.test(line)) {
          report('UNHANDLED_ERROR', { line });
        }
      });

      position = stats.size;
    });
  }

  fs.watch(filePath, { persistent: true }, () => {
    readNew();
  });

  readNew();
}

function monitorSessionChanges() {
  setInterval(() => {
    const latest = getLatestSessionDir();
    if (!latest) return;

    if (latest !== currentSessionDir) {
      currentSessionDir = latest;
      report('NEW_SESSION_DETECTED', { session: latest });

      const exthost = path.join(latest, 'exthost.log');
      if (fs.existsSync(exthost)) {
        watchLogFile(exthost);
      }
    }
  }, 2000);
}

function monitorExtensionHost() {
  setInterval(() => {
    const ps = spawn('pgrep', ['-f', 'extensionHost']);

    let output = '';

    ps.stdout.on('data', d => output += d);

    ps.on('close', () => {
      const pids = output.trim().split('\n').filter(Boolean);

      if (pids.length === 0) {
        report('EXTENSION_HOST_NOT_RUNNING', {});
        return;
      }

      if (!extensionHostPid) {
        extensionHostPid = pids[0];
        return;
      }

      if (pids[0] !== extensionHostPid) {
        report('EXTENSION_HOST_RESTARTED', {
          oldPid: extensionHostPid,
          newPid: pids[0]
        });
        extensionHostPid = pids[0];
      }
    });

  }, 2000);
}

function main() {
  if (!fs.existsSync(LOG_ROOT)) {
    console.error('VS Code log directory not found.');
    process.exit(1);
  }

  report('MONITOR_STARTED', {});

  monitorSessionChanges();
  monitorExtensionHost();
}

main();
