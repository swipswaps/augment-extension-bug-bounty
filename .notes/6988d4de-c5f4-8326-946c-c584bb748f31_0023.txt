export function activate(context: vscode.ExtensionContext) {

    output.show(true);

    output.appendLine(`=== Hidden Terminal Watchdog Activated ===`);
    output.appendLine(`Date: ${new Date().toISOString()}`);

    // Track all terminals
    const trackedTerminals = new Set<vscode.Terminal>();

    // Monitor newly created terminals
    vscode.window.onDidOpenTerminal((term) => {

        output.appendLine(`[INFO] Terminal opened: ${term.name}`);

    // Monitor closed terminals
    vscode.window.onDidCloseTerminal((term) => {
        trackedTerminals.delete(term);
        output.appendLine(`[INFO] Terminal closed: ${term.name}`);

    const interval = setInterval(() => {
        exec(

                const pids = stdout.split(/\s+/).filter(Boolean);
                output.appendLine(`[INFO] Detected ${pids.length} active processes: ${pids.join(', ')}`);
                if (pids.length > 20) { // arbitrary threshold
                    output.appendLine('[WARN] Too many active terminals, consider cleanup.');
                }

import * as vscode from 'vscode';
import { exec } from 'child_process';
import * as os from 'os';

export function activate(context: vscode.ExtensionContext) {
    const output = vscode.window.createOutputChannel('Hidden Terminal Watchdog');
    output.show(true);

    const startTime = new Date();
    output.appendLine(`=== Hidden Terminal Watchdog Activated ===`);
    output.appendLine(`Start Time: ${startTime.toISOString()}`);
    output.appendLine(`User: ${os.userInfo().username}`);
    output.appendLine(`VS Code PID: ${process.pid}`);

    // Track all VS Code integrated terminals
    const trackedTerminals = new Set<vscode.Terminal>();

    // Observe terminal creation
    vscode.window.onDidOpenTerminal((term) => {
        trackedTerminals.add(term);
        output.appendLine(`[INFO] Terminal opened: ${term.name}`);
    });

    // Observe terminal closure
    vscode.window.onDidCloseTerminal((term) => {
        trackedTerminals.delete(term);
        output.appendLine(`[INFO] Terminal closed: ${term.name}`);
    });

    // Internal function: Detect hidden terminals and extension hosts
    function detectHiddenTerminals() {
        const pattern = 'code.*--ms-enable-electron-run-as-node|extensionHost|bash|zsh|fish';
        exec(`pgrep -u $USER -f "${pattern}"`, (err, stdout, stderr) => {
            if (err) return; // No matches found
            const pids = stdout.split(/\s+/).filter(Boolean);

            if (pids.length > 0) {
                output.appendLine(`[WARN] Detected ${pids.length} hidden terminals/processes: ${pids.join(', ')}`);

                // Optional: log command lines
                pids.forEach((pid) => {
                    exec(`tr '\\0' ' ' < /proc/${pid}/cmdline`, (cmdErr, cmdOut) => {
                        if (!cmdErr && cmdOut.trim()) {
                            output.appendLine(`  PID ${pid}: ${cmdOut.trim()}`);
                        } else {
                            output.appendLine(`  PID ${pid}: [cmdline unavailable]`);
                        }
                    });
                });
            }
        });
    }

    // Monitor hidden terminals every 3 seconds asynchronously
    const interval = setInterval(detectHiddenTerminals, 3000);

    // Register command to force cleanup via existing bash script
    const cleanupCmd = vscode.commands.registerCommand('watchdog.cleanup', () => {
        exec('bash ./kill-hidden-terminals.sh --force', (err, stdout, stderr) => {
            output.appendLine(`=== Force Cleanup Output ===`);
            if (stdout.trim()) output.appendLine(stdout.trim());
            if (err) output.appendLine(`[ERROR] ${stderr.trim()}`);
            output.appendLine(`=== Cleanup Complete ===`);
        });
    });

    // Save subscriptions
    context.subscriptions.push(cleanupCmd, { dispose: () => clearInterval(interval) });

    // Log periodic heartbeat
    const heartbeat = setInterval(() => {
        output.appendLine(`[HEARTBEAT] Watchdog active at ${new Date().toISOString()}. Tracked terminals: ${trackedTerminals.size}`);
    }, 15000);
    context.subscriptions.push({ dispose: () => clearInterval(heartbeat) });
}
