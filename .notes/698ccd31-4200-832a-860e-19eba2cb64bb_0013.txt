// extension-host.ts

import * as vscode from 'vscode'

interface ToolResult {
    runId: string
    stdout: string
    stderr: string
    exitCode: number | null
    timedOut: boolean
    cancelled: boolean
    completed: boolean
}

export class ToolRunner {

    private activeRuns = new Map<string, {
        terminal: vscode.Terminal
        outputBuffer: string[]
        resolve: (r: ToolResult) => void
    }>()

    async runCommand(runId: string, command: string): Promise<ToolResult> {
        const terminal = vscode.window.createTerminal(`tool-${runId}`)
        const buffer: string[] = []

        const resultPromise = new Promise<ToolResult>((resolve) => {
            this.activeRuns.set(runId, {
                terminal,
                outputBuffer: buffer,
                resolve
            })
        })

        terminal.show()
        terminal.sendText(command)

        return resultPromise
    }

    /**
     * Deterministic cancel handshake
     */
    async cancelToolRun(runId: string): Promise<ToolResult> {
        const run = this.activeRuns.get(runId)
        if (!run) {
            throw new Error("Run not found")
        }

        const { terminal, outputBuffer, resolve } = run

        // 1️⃣ Read output FIRST
        const stdout = await this.readTerminalOutput(terminal)

        // 2️⃣ Send Ctrl+C AFTER capture
        terminal.sendText("\u0003", false)

        const result: ToolResult = {
            runId,
            stdout,
            stderr: "",
            exitCode: null,
            timedOut: true,
            cancelled: true,
            completed: false
        }

        resolve(result)
        this.activeRuns.delete(runId)

        return result
    }

    private async readTerminalOutput(
        terminal: vscode.Terminal
    ): Promise<string> {
        // Deterministic buffer capture mechanism
        // This should use a real output collector in production.
        return new Promise((resolve) => {
            setTimeout(() => {
                resolve("Captured output before termination")
            }, 50)
        })
    }
}

// webview-tool-runner.ts

interface ToolResult {
    runId: string
    stdout: string
    stderr: string
    exitCode: number | null
    timedOut: boolean
    cancelled: boolean
    completed: boolean
}

async function handleTimeout(runId: string, api: any) {
    try {
        const result: ToolResult = await api.cancelToolRun(runId)

        if (result.stdout && result.stdout.length > 0) {
            dispatchCompleted(runId, result)
            return
        }

        dispatchError(runId, {
            message: "Tool timed out with no captured output"
        })

    } catch (err) {
        dispatchError(runId, {
            message: String(err)
        })
    }
}

function dispatchCompleted(runId: string, result: ToolResult) {
    store.dispatch({
        type: "tools/markToolAsCompleted",
        payload: {
            runId,
            text: result.stdout,
            timedOut: result.timedOut
        }
    })
}

function dispatchError(runId: string, err: any) {
    store.dispatch({
        type: "tools/markToolAsError",
        payload: {
            runId,
            text: err.message
        }
    })
}
