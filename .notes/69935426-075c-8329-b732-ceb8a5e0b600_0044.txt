/**
 * DeterministicRunner.ts
 *
 * This module replaces ALL direct shell execution.
 *
 * Design goals:
 *  - Always visible execution
 *  - Live stdout + stderr streaming
 *  - Tee logging to disk
 *  - Cancellation-resilient
 *  - Idempotent retry safe
 *  - No hidden terminals
 *
 * Based on:
 *  - Node.js child_process.spawn streaming model
 *  - VS Code Extension API Terminal visibility model
 */

import { spawn } from "child_process";
import * as fs from "fs";
import * as path from "path";

export interface RunOptions {
    cwd?: string;
    env?: NodeJS.ProcessEnv;
    logFile?: string;
}

export class DeterministicRunner {

    static async run(command: string, options: RunOptions = {}): Promise<number> {

        const logPath = options.logFile || path.join(process.cwd(), "augment.log");

        // Ensure log file exists
        fs.appendFileSync(logPath, `\n\n===== EXEC: ${command} =====\n`);

        return new Promise((resolve, reject) => {

            /**
             * IMPORTANT:
             * Using spawn instead of exec to avoid:
             * - 200KB buffer truncation
             * - Silent kill on large output
             */
            const child = spawn(command, {
                shell: true,
                cwd: options.cwd || process.cwd(),
                env: options.env || process.env,
                stdio: ["inherit", "pipe", "pipe"]
            });

            child.stdout.on("data", (data) => {
                const output = data.toString();

                process.stdout.write(output);           // Visible TTY output
                fs.appendFileSync(logPath, output);     // Tee to file
            });

            child.stderr.on("data", (data) => {
                const errorOutput = data.toString();

                process.stderr.write(errorOutput);      // Visible TTY error
                fs.appendFileSync(logPath, errorOutput);
            });

            child.on("close", (code) => {

                fs.appendFileSync(
                    logPath,
                    `\n===== EXIT CODE: ${code} =====\n`
                );

                /**
                 * CRITICAL RULE:
                 * Do NOT treat null or undefined exit as cancellation.
                 * Only non-zero numeric exit is failure.
                 */
                if (code === 0) {
                    resolve(0);
                } else {
                    reject(new Error(`Command exited with code ${code}`));
                }
            });

            child.on("error", (err) => {
                fs.appendFileSync(logPath, `\nPROCESS ERROR: ${err.message}\n`);
                reject(err);
            });
        });
    }
}

/**
 * VisibleTerminal.ts
 *
 * Ensures:
 *  - No hidden terminals
 *  - Terminal always shown
 *  - Terminal reuse
 */

import * as vscode from "vscode";

export class VisibleTerminal {

    private static terminal: vscode.Terminal | null = null;

    static get(): vscode.Terminal {

        if (!this.terminal) {
            this.terminal = vscode.window.createTerminal({
                name: "Augment Deterministic Terminal",
                hideFromUser: false,      // CRITICAL: never hidden
                isTransient: false
            });
        }

        this.terminal.show(true); // force reveal
        return this.terminal;
    }

    static send(command: string) {
        const term = this.get();
        term.sendText(command, true);
    }
}

/**
 * CancellationGuard.ts
 *
 * Detects false cancellation artifacts such as:
 *   "Cancelled by user."
 *   empty output
 *   partial output
 */

export function isSpuriousCancellation(output: string): boolean {

    if (!output) return true;

    if (output.includes("Cancelled by user.")) return true;
    if (output.includes("tool call was cancelled")) return true;

    return false;
}
