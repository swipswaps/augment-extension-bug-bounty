###############################################################################
# RETRY-ENFORCING COMPLIANCE DRIVER
#
# Governing Authority:
#   mandatory-rules-v6.6.md
#   :contentReference[oaicite:1]{index=1}
#
# Execution Mode Enforcement:
#   - RULE 4: Mode Locking — No planning, no deferral, direct execution.
#
# Emission Gate:
#   - RULE 0: No artifact may be emitted unless checks pass.
#   This script self-validates output before declaring success.
#
# Terminal Discipline:
#   - RULE 8: launch-process (wait=true) required pattern.
#   - RULE 9: Forbidden tools avoided (no read-process, no list-processes).
#   - RULE 22: Single terminal session. No recursive spawn.
#
# Retry Enforcement:
#   - RULE 15: Retry only if:
#       * <output> empty
#       * START without END
#       * explicit ERROR
#
# Determinism:
#   - RULE 12: Fixed retry count. No randomness.
#
# Zero-Hang Guarantee:
#   - RULE 15: Hard bounded retry loop.
#
# Environment Declaration (RULE 20):
#   Requires:
#       - bash >= 4
#       - .notes directory writable
#       - .augment/scripts/terminal-watchdog.sh
#       - .augment/scripts/pre-response-check.sh
#
###############################################################################

set -euo pipefail

MAX_RETRIES=3
RETRY_DELAY_SECONDS=2
ATTEMPT=1

TIMESTAMP="$(date +%Y%m%d-%H%M%S)"
LOG_DIR=".notes"
LOG_FILE="${LOG_DIR}/terminal-${TIMESTAMP}.log"

mkdir -p "${LOG_DIR}"

###############################################################################
# FUNCTION: execute_target
#
# RULE 22 compliance:
#   Entire retry loop runs inside this single shell process.
#   No additional terminals spawned.
###############################################################################
execute_target() {

    ###########################################################################
    # Replace the command below with the actual instrumented command.
    # Must produce <output>...</output> markers if compliant.
    #
    # Example placeholder command is deterministic and safe.
    ###########################################################################

    bash -c '
        echo "<output>"
        echo "Sample execution payload"
        echo "</output>"
    '
}

###############################################################################
# FUNCTION: validate_output
#
# RULE 15 enforcement:
#   Retry if:
#       - output empty
#       - START without END
#       - explicit ERROR detected
###############################################################################
validate_output() {

    local file="$1"

    if [[ ! -s "$file" ]]; then
        return 1
    fi

    if ! grep -q "<output>" "$file"; then
        return 1
    fi

    if ! grep -q "</output>" "$file"; then
        return 1
    fi

    if grep -q "ERROR" "$file"; then
        return 1
    fi

    return 0
}

###############################################################################
# MAIN RETRY LOOP
#
# RULE 12: Deterministic retry bound.
# RULE 15: Hard stop after MAX_RETRIES.
###############################################################################
while [[ "$ATTEMPT" -le "$MAX_RETRIES" ]]; do

    echo "=== ATTEMPT ${ATTEMPT} ===" | tee -a "$LOG_FILE"

    ###########################################################################
    # tee is permitted here because:
    #   - It is within same terminal session.
    #   - It does NOT spawn separate launch-process calls.
    ###########################################################################
    execute_target 2>&1 | tee -a "$LOG_FILE"

    if validate_output "$LOG_FILE"; then
        echo "VALID OUTPUT DETECTED" | tee -a "$LOG_FILE"
        SUCCESS=1
        break
    else
        echo "INVALID OR EMPTY OUTPUT — RETRYING" | tee -a "$LOG_FILE"
        SUCCESS=0
        sleep "$RETRY_DELAY_SECONDS"
    fi

    ATTEMPT=$((ATTEMPT + 1))

done

###############################################################################
# POST-EXECUTION WATCHDOGS
#
# RULE 8 + RULE 9 compliance:
#   Must run watchdog and pre-response-check before declaring success.
###############################################################################
if [[ -x ".augment/scripts/terminal-watchdog.sh" ]]; then
    .augment/scripts/terminal-watchdog.sh | tee -a "$LOG_FILE"
fi

if [[ -x ".augment/scripts/pre-response-check.sh" ]]; then
    .augment/scripts/pre-response-check.sh | tee -a "$LOG_FILE"
fi

###############################################################################
# FINAL STATUS
#
# RULE 0: No success without validated evidence.
###############################################################################
if [[ "${SUCCESS:-0}" -eq 1 ]]; then
    echo "COMPLIANCE DRIVER SUCCESS" | tee -a "$LOG_FILE"
    exit 0
else
    echo "COMPLIANCE DRIVER FAILURE AFTER ${MAX_RETRIES} ATTEMPTS" | tee -a "$LOG_FILE"
    exit 1
fi
