5. Working Reference — Robust Profile Detection (Node)

This replaces fragile grep/awk shell parsing entirely.

profileDetector.js
import fs from 'fs/promises'
import fsSync from 'fs'
import path from 'path'
import os from 'os'

const BASES = [
  path.join(os.homedir(), '.mozilla/firefox'),
  path.join(os.homedir(), '.var/app/org.mozilla.firefox/.mozilla/firefox'),
  path.join(os.homedir(), 'snap/firefox/common/.mozilla/firefox')
]

export async function detectProfiles() {
  const results = []

  for (const base of BASES) {
    const ini = path.join(base, 'profiles.ini')
    if (!fsSync.existsSync(ini)) continue

    const content = await fs.readFile(ini, 'utf8')
    const lines = content.split('\n')

    let current = {}

    for (const line of lines) {
      if (line.startsWith('[Profile')) current = {}

      if (line.startsWith('Name=')) current.name = line.split('=')[1]
      if (line.startsWith('Path=')) current.path = line.split('=')[1]
      if (line.startsWith('Default=1')) current.default = true

      if (current.path && current.name) {
        const full = path.join(base, current.path)
        if (fsSync.existsSync(full)) {
          results.push({
            ...current,
            fullPath: full,
            prefsExists: fsSync.existsSync(path.join(full, 'prefs.js'))
          })
        }
      }
    }
  }

  return results
}


This matches how serious profile managers enumerate profiles.

6. Working Reference — Semantic Pref Diff

Text diff is not enough.

prefDiff.js
export function parsePrefs(content) {
  const map = {}
  const lines = content.split('\n')

  for (const line of lines) {
    const match = line.match(/user_pref\("([^"]+)",\s*(.+)\);/)
    if (match) map[match[1]] = match[2]
  }
  return map
}

export function diffPrefs(oldContent, newContent) {
  const oldMap = parsePrefs(oldContent)
  const newMap = parsePrefs(newContent)

  const changes = []

  for (const key of new Set([...Object.keys(oldMap), ...Object.keys(newMap)])) {
    if (oldMap[key] !== newMap[key]) {
      changes.push({
        key,
        before: oldMap[key],
        after: newMap[key]
      })
    }
  }

  return changes
}


This is cutting-edge for config UX because:

user sees actual pref changes

not noisy text diff

7. Working Reference — Safe Apply Engine
applyConfig.js
import fs from 'fs/promises'
import fsSync from 'fs'
import path from 'path'

const MAX_BACKUPS = 5

export async function rotateBackups(file) {
  for (let i = MAX_BACKUPS - 1; i >= 1; i--) {
    const oldF = `${file}.backup.${i}`
    const newF = `${file}.backup.${i + 1}`
    if (fsSync.existsSync(oldF)) await fs.rename(oldF, newF)
  }
  if (fsSync.existsSync(file)) {
    await fs.copyFile(file, `${file}.backup.1`)
  }
}

export async function safeWriteUserJS(profileDir, content) {
  const userJS = path.join(profileDir, 'user.js')

  await rotateBackups(userJS)
  await fs.writeFile(userJS, content, 'utf8')
}

What “Self-Healing” Means (Precisely)

A self-healing configuration system must:

Detect failure signals

Know the last known good state

Restore automatically

Never loop endlessly

Tell the user exactly what happened

For Firefox tuning, the real failure we care about is:

Firefox fails to start, crashes repeatedly, or ignores prefs after a config change.

1. Failure Signals (What We Can Reliably Detect)

We do NOT guess. We rely on observable signals.

Reliable Signals on Linux

Cutting-edge best practice is to combine two independent checks:

Startup health check

Firefox process starts and remains alive for N seconds

Preference acceptance check

Firefox writes to prefs.js after startup

If either fails → config is unsafe.

2. Last Known Good (LKG) Definition

A config is Last Known Good if:

Firefox started successfully

Firefox ran for ≥ 10 seconds

prefs.js timestamp updated

no crash loop detected

We explicitly mark LKG, not infer it later.

3. Architecture (Non-Negotiable)
Apply config
   ↓
Mark as "pending"
   ↓
Run health check
   ↓
✔ success → promote to LKG
✖ failure → rollback to LKG


No other flow is acceptable in production tools.

6. Working Code — Self-Healing Engine (Node)

This is drop-in usable.

selfHeal.js
import fs from 'fs/promises'
import fsSync from 'fs'
import path from 'path'
import os from 'os'
import { execFile } from 'child_process'
import { promisify } from 'util'

const execFileAsync = promisify(execFile)

const STATE_DIR = path.join(os.homedir(), '.cache/firefox-hud')
const STATE_FILE = path.join(STATE_DIR, 'state.json')
const LKG_FILE = path.join(STATE_DIR, 'last-known-good.js')

await fs.mkdir(STATE_DIR, { recursive: true })

async function readState() {
  if (!fsSync.existsSync(STATE_FILE)) {
    return { status: 'stable', attempts: 0 }
  }
  return JSON.parse(await fs.readFile(STATE_FILE, 'utf8'))
}

async function writeState(state) {
  await fs.writeFile(STATE_FILE, JSON.stringify(state, null, 2))
}

async function firefoxHealthy(timeoutMs = 10000) {
  try {
    const { stdout } = await execFileAsync(
      'bash',
      ['-c', `firefox >/dev/null 2>&1 & sleep 2 && pgrep -x firefox`],
      { timeout: timeoutMs }
    )
    return stdout.trim().length > 0
  } catch {
    return false
  }
}

export async function promoteLastKnownGood(userJsPath) {
  const content = await fs.readFile(userJsPath, 'utf8')
  await fs.writeFile(LKG_FILE, content)
  await writeState({ status: 'stable', attempts: 0, lastGoodTimestamp: Date.now() })
}

export async function rollback(userJsPath) {
  if (!fsSync.existsSync(LKG_FILE)) {
    throw new Error('No last known good config available')
  }

  const content = await fs.readFile(LKG_FILE, 'utf8')
  await fs.writeFile(userJsPath, content)

  await writeState({ status: 'rolled_back', attempts: 0 })
}

export async function selfHealIfNeeded(userJsPath) {
  const state = await readState()

  if (state.status !== 'pending') return

  if (state.attempts >= 2) {
    await writeState({ status: 'locked', attempts: state.attempts })
    return
  }

  const healthy = await firefoxHealthy()

  if (healthy) {
    await promoteLastKnownGood(userJsPath)
  } else {
    await rollback(userJsPath)
    await writeState({
      status: 'rolled_back',
      attempts: state.attempts + 1
    })
  }
}
