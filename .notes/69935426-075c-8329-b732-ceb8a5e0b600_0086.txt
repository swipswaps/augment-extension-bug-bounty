// HELPER: Extract code blocks marked with ```[js|bash]
function extractCodeBlocks(content) {
  const blocks = [];
  const regex = /```(js|bash)\s*([\s\S]*?)```/g;
  let match;
  while ((match = regex.exec(content)) !== null) {
    const lang = match[1];
    const code = match[2];
    blocks.push({ lang, code });
  }
  return blocks;
}
