// FILE: src/core/ExecBanEnforcer.ts
//
// PURPOSE:
// This file programmatically prevents the extension from ever using exec() again.
//
// WHY:
// exec() buffers output (200KB default in Node.js).
// When exceeded, output is truncated.
// When the extension host reloads, exec promises can hang.
// This is documented in Node.js child_process docs.
//
// COMPLIANCE RULE:
// If exec() appears anywhere in the codebase, the extension must refuse activation.
//
// IMPLEMENTATION:
// We scan our own source files at activation time.
// If "exec(" is detected, throw a hard error.
//
// This prevents regression and enforces compliance automatically.

import * as fs from "fs";
import * as path from "path";

export class ExecBanEnforcer {

    static enforce(workspaceRoot: string) {

        const srcDir = path.join(workspaceRoot, "src");

        function scan(dir: string) {
            const entries = fs.readdirSync(dir);

            for (const entry of entries) {
                const full = path.join(dir, entry);
                const stat = fs.statSync(full);

                if (stat.isDirectory()) {
                    scan(full);
                } else if (full.endsWith(".ts")) {

                    const content = fs.readFileSync(full, "utf8");

                    if (content.includes("exec(")) {
                        throw new Error(
                            `COMPLIANCE FAILURE: exec() detected in ${full}`
                        );
                    }
                }
            }
        }

        scan(srcDir);
    }
}

// FILE: src/core/CodeInspector.ts

// PURPOSE:
// Replace shell-based grep pipelines with deterministic TypeScript scanning.
// This avoids interactive terminal blocking and buffering issues.

import * as fs from "fs";

export class CodeInspector {

    static findExecOccurrences(filePath: string): number {

        const content = fs.readFileSync(filePath, "utf8");

        const matches = content.match(/exec\(/g);

        return matches ? matches.length : 0;
    }

    static reportExecOccurrences(filePath: string) {

        const content = fs.readFileSync(filePath, "utf8");

        const lines = content.split("\n");

        lines.forEach((line, index) => {
            if (line.includes("exec(")) {
                console.log(
                    `exec() found at line ${index + 1}: ${line.trim()}`
                );
            }
        });
    }
}

// FILE: src/core/StructuredLogger.ts
//
// PURPOSE:
// Deterministic logging without shell tee pipelines.
// Avoids blocking and terminal buffering.
//
// WHY:
// tee relies on shell and interactive pipelines.
// That causes blocking in VS Code integrated terminals.

import * as fs from "fs";

export class StructuredLogger {

    private static logFile = `.augment-structured-${Date.now()}.log`;

    static log(message: string) {

        const line = `[${new Date().toISOString()}] ${message}\n`;

        process.stdout.write(line);
        fs.appendFileSync(this.logFile, line);
    }

    static getLogFilePath(): string {
        return this.logFile;
    }
}
