const { spawn } = require("child_process");
const fs = require("fs");

function startResumableDownload(id, url) {

  const outputFile = path.join(os.tmpdir(), `youtube-${id}.mp4`);

  const args = [
    "-f", "18/bv*[ext=mp4][protocol!=m3u8]+ba[ext=m4a][protocol!=m3u8]/b[ext=mp4][protocol!=m3u8]",
    "--merge-output-format", "mp4",

    /**
     * KEY: Resume support
     * --continue resumes partial files
     */
    "--continue",

    /**
     * KEY: Emit structured progress lines
     */
    "--newline",

    "-o", outputFile,
    url
  ];

  const proc = spawn("yt-dlp", args);

  activeDownloads.set(id, {
    process: proc,
    progress: 0,
    filePath: outputFile,
    lastUpdate: Date.now()
  });

  proc.stdout.on("data", data => {
    const text = data.toString();

    /**
     * yt-dlp progress lines look like:
     * [download]  23.4% of 12.34MiB at 1.23MiB/s ETA 00:10
     */

    const match = text.match(/(\d+\.\d+)%/);
    if (match) {
      const percent = parseFloat(match[1]);

      const entry = activeDownloads.get(id);
      if (entry) {
        entry.progress = percent;
        entry.lastUpdate = Date.now();
      }
    }
  });

  proc.on("close", () => {
    activeDownloads.delete(id);
  });

  return outputFile;
}

  const { id } = req.body;

  const entry = activeDownloads.get(id);

  if (!entry) {
    return res.status(404).json({ error: "Download not found" });
  }

  entry.process.kill("SIGTERM");

  activeDownloads.delete(id);

  return res.json({ success: true, cancelled: id });
});

STEP 2 — Adaptive Launch Threshold

Replace static 5MB with bitrate-aware logic:

/**
 * DYNAMIC PLAYBACK THRESHOLD
 *
 * Instead of arbitrary 5MB,
 * use estimated bitrate from progress line.
 *
 * If bitrate not parsed yet,
 * fallback to 8MB.
 */

let estimatedBitrate = null;

proc.stdout.on("data", data => {
  const text = data.toString();

  const percentMatch = text.match(/(\d+\.\d+)%/);
  const speedMatch = text.match(/at\s+([\d.]+)(MiB|KiB)/);

  if (speedMatch) {
    const value = parseFloat(speedMatch[1]);
    const unit = speedMatch[2];
    estimatedBitrate = unit === "MiB"
      ? value * 1024 * 1024
      : value * 1024;
  }

  if (percentMatch) {
    const entry = activeDownloads.get(id);
    if (entry) {
      entry.progress = parseFloat(percentMatch[1]);
      entry.lastUpdate = Date.now();
    }
  }
});


Then compute threshold:

function computeLaunchThreshold() {
  if (!estimatedBitrate) return 8 * 1024 * 1024; // 8MB fallback

  /**
   * Require at least 10 seconds of buffered video.
   */
  return estimatedBitrate * 10;
}
