const EVENT_LOOP_DRIFT_THRESHOLD = 4000;

let lastHeartbeat = Date.now();

function heartbeat() {

}

function monitorEventLoop() {
    let lastTick = Date.now();

    setInterval(() => {
        const now = Date.now();
        const drift = now - lastTick - HEARTBEAT_INTERVAL;

        if (drift > EVENT_LOOP_DRIFT_THRESHOLD) {
            log(`CRITICAL | Event loop stall detected | drift=${drift}ms`);
            attemptRecovery("event-loop-stall");
        }

        lastTick = now;
    }, HEARTBEAT_INTERVAL);
}

function monitorTerminals() {
    vscode.window.onDidOpenTerminal(() => {
        const count = vscode.window.terminals.length;

        if (count > MAX_TERMINALS) {
            log(`WARNING | Terminal overload detected | count=${count}`);
            cleanupTerminals();
        }
    });

    vscode.window.onDidCloseTerminal(() => {
        log(`INFO | Terminal closed | count=${vscode.window.terminals.length}`);
    });
}

function cleanupTerminals() {
    vscode.window.terminals.forEach(term => {
        if (!term.name.includes("persistent")) {
            term.dispose();
        }
    });

    log("ACTION | Non-persistent terminals disposed.");
}

function monitorProcesses() {
    setInterval(() => {
        exec("ps -eo pid,comm | grep -E 'node|npm' | grep -v grep", (err, stdout) => {
            if (err || !stdout) return;

            const processes = stdout.split('\n').filter(Boolean);

            if (processes.length > MAX_NODE_PROCESSES) {
                log(`CRITICAL | Node process overload | count=${processes.length}`);
                attemptRecovery("node-overload");
            }
        });
    }, PROCESS_SCAN_INTERVAL);
}

function monitorCancellationPatterns() {
    vscode.window.onDidWriteTerminalData(event => {
        if (event.data.includes("Cancelled by user")) {
            cancellationEvents++;
            log(`DETECTED | Cancellation pattern observed | total=${cancellationEvents}`);
            attemptRecovery("cancellation-pattern");
        }
    });
}

        }
    });
}

export function handleToolResult(rawOutput: string): string {
    const cancelled = rawOutput.includes("<error>Cancelled by user.</error>");
    const outputMatch = rawOutput.match(/<output>([\s\S]*?)<\/output>/);

    if (cancelled) {
        console.log("Cancellation detected. Overriding stop condition.");
    }

    if (outputMatch) {
        const output = outputMatch[1];
        console.log("Recovered output:\n", output);
        return output;
    }

    console.log("No output section found. Initiating recovery protocol.");
    runRecoveryProtocol();
    return "";
}

async function runRecoveryProtocol() {
    const commands = [
        "ps aux",
        "ls -la",
        "tail -n 50 ~/.bash_history || true",
        "ss -tulnp || netstat -tulnp || true"

    for (const cmd of commands) {
        try {

            logEvidence(cmd, stdout);

            console.error(`Recovery command failed: ${cmd}`, err);
        }
    }
}

function logEvidence(command: string, output: string) {
    const channel = getChannel();
    channel.appendLine(`\n=== ${command} ===\n${output}`);
}

async function runTerminalHygiene() {
    const config = vscode.workspace.getConfiguration("augmentGuard");
    const maxTerminals = config.get<number>("maxTerminals") || 20;

    const terminals = vscode.window.terminals;

    if (terminals.length > maxTerminals) {
        vscode.window.showWarningMessage(
            `Terminal count (${terminals.length}) exceeds threshold (${maxTerminals}). Cleaning up.`

        terminals.slice(maxTerminals).forEach(t => t.dispose());

async function killRunawayProcesses() {

        }
    }

import * as vscode from "vscode";
import { spawn } from "child_process";

export interface TeeOptions {
    cwd?: string;
    env?: NodeJS.ProcessEnv;
}

export function runWithTee(
    command: string,
    args: string[],
    options: TeeOptions = {}
): Promise<number> {

    return new Promise((resolve) => {

        const terminal = getOrCreateTerminal();
        const outputChannel = getOrCreateChannel();

        terminal.show(true);

        const child = spawn(command, args, {
            cwd: options.cwd || process.cwd(),
            env: options.env || process.env,
            shell: true
        });

        terminal.sendText(`\n# Executing: ${command} ${args.join(" ")}\n`);

        child.stdout.on("data", (data) => {
            const text = data.toString();
            terminal.sendText(text, false);
            outputChannel.append(text);
        });

        child.stderr.on("data", (data) => {
            const text = data.toString();
            terminal.sendText(text, false);
            outputChannel.append(text);
        });

        child.on("close", (code) => {
            terminal.sendText(`\n# Process exited with code ${code}\n`);
            outputChannel.appendLine(`Process exited with code ${code}`);
            resolve(code ?? 0);
        });

        child.on("error", (err) => {
            terminal.sendText(`\n# Spawn error: ${err.message}\n`);
            outputChannel.appendLine(`Spawn error: ${err.message}`);
            resolve(1);
        });
    });
}

let terminalInstance: vscode.Terminal | undefined;
let outputChannelInstance: vscode.OutputChannel | undefined;

function getOrCreateTerminal(): vscode.Terminal {
    if (!terminalInstance) {
        terminalInstance = vscode.window.createTerminal("Augment Tee");
    }
    return terminalInstance;
}

function getOrCreateChannel(): vscode.OutputChannel {
    if (!outputChannelInstance) {
        outputChannelInstance = vscode.window.createOutputChannel("Augment Tee Log");
    }
    return outputChannelInstance;
}
