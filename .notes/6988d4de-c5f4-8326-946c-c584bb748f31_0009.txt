FILE 1 — augment_safe_exec.sh

This is the only script Augment / VS Code should run.

It wraps any command and guarantees output is captured atomically.

#!/usr/bin/env bash
# =============================================================================
# augment_safe_exec.sh
# =============================================================================
# PURPOSE:
#   Execute a command in a way that is immune to:
#     - VS Code terminal truncation
#     - Stream reader timeouts
#     - cleanupTerminal() deletion races
#     - Shell Integration suppression
#
# DESIGN PRINCIPLE:
#   Never trust terminal streams.
#   Always persist output to disk FIRST.
#
# USAGE:
#   ./augment_safe_exec.sh <output_dir> <command...>
#
# EXAMPLE:
#   ./augment_safe_exec.sh /tmp/augment_out ls -la /
#
# =============================================================================

set -Eeuo pipefail

# -------------------------------
# Validate input
# -------------------------------
if [[ $# -lt 2 ]]; then
  echo "USAGE ERROR:"
  echo "  augment_safe_exec.sh <output_dir> <command...>"
  exit 64
fi

OUTPUT_DIR="$1"
shift

mkdir -p "$OUTPUT_DIR"

# -------------------------------
# Stable filenames (never deleted early)
# -------------------------------
STDOUT_FILE="$OUTPUT_DIR/stdout.log"
STDERR_FILE="$OUTPUT_DIR/stderr.log"
META_FILE="$OUTPUT_DIR/meta.json"

# -------------------------------
# Clear previous contents safely
# -------------------------------
: > "$STDOUT_FILE"
: > "$STDERR_FILE"

START_TS="$(date --iso-8601=seconds)"

# -------------------------------
# Execute command with full capture
# -------------------------------
# NOTE:
#   - stdbuf disables libc buffering
#   - tee writes output immediately
#   - exit code preserved via PIPESTATUS
#
(
  stdbuf -oL -eL "$@"
) \
  > >(tee -a "$STDOUT_FILE") \
  2> >(tee -a "$STDERR_FILE" >&2)

EXIT_CODE="${PIPESTATUS[0]}"

END_TS="$(date --iso-8601=seconds)"

# -------------------------------
# Emit metadata LAST (atomic)
# -------------------------------
cat > "$META_FILE" <<EOF
{
  "start_time": "$START_TS",
  "end_time": "$END_TS",
  "exit_code": $EXIT_CODE,
  "command": "$(printf '%q ' "$@")"
}
EOF

# -------------------------------
# Terminal output (small + safe)
# -------------------------------
# This is intentionally minimal so truncation
# does NOT matter.
#
echo "OUTPUT_DIR=$OUTPUT_DIR"
echo "EXIT_CODE=$EXIT_CODE"

exit "$EXIT_CODE"
