const { execFile } = require("child_process");
const path = require("path");
const fs = require("fs");
const os = require("os");

function downloadYouTube(url) {
  return new Promise((resolve, reject) => {
    const tempDir = os.tmpdir();
    const outputTemplate = path.join(tempDir, "youtube-%(id)s.%(ext)s");

    const args = [
      "-f", "bv*[ext=mp4]+ba[ext=m4a]/b[ext=mp4]",
      "--merge-output-format", "mp4",
      "--no-part",
      "--retries", "10",
      "--fragment-retries", "10",
      "--concurrent-fragments", "5",
      "-o", outputTemplate,
      url
    ];

    const proc = execFile("yt-dlp", args);

    let stderr = "";
    proc.stderr.on("data", d => stderr += d.toString());

    proc.on("close", code => {
      if (code !== 0) {
        return reject(new Error(stderr));
      }

      // Find most recent downloaded file
      const files = fs.readdirSync(tempDir)
        .filter(f => f.startsWith("youtube-") && f.endsWith(".mp4"))
        .map(f => ({
          file: f,
          time: fs.statSync(path.join(tempDir, f)).mtime.getTime()
        }))
        .sort((a,b) => b.time - a.time);

      if (!files.length) {
        return reject(new Error("No MP4 file found after download"));
      }

      resolve(path.join(tempDir, files[0].file));
    });
  });
}

const { execFile } = require("child_process");
const util = require("util");
const execFileAsync = util.promisify(execFile);

async function verifyBinaryExists(binary) {
  try {
    await execFileAsync("which", [binary]);
    return true;
  } catch {
    return false;
  }
}

/**
 * DOWNLOAD YOUTUBE VIDEO SAFELY
 *
 * This function:
 *  - Forces progressive MP4 format
 *  - Merges audio + video using ffmpeg
 *  - Avoids HLS manifest playback
 *  - Retries fragments
 *  - Logs stderr for deterministic debugging
 */

const fs = require("fs");
const path = require("path");
const os = require("os");

async function downloadYouTubeVideo(url) {

  // Ensure required binaries exist
  if (!(await verifyBinaryExists("yt-dlp"))) {
    throw new Error("yt-dlp not installed");
  }

  if (!(await verifyBinaryExists("ffmpeg"))) {
    throw new Error("ffmpeg not installed");
  }

  await verifyYtDlpVersion();
  await verifyFfmpeg();

  const tempDir = os.tmpdir();

  /**
   * We use %(id)s so file name is deterministic per video.
   * This prevents multiple timestamp-based duplicates.
   */
  const outputTemplate = path.join(tempDir, "youtube-%(id)s.%(ext)s");

  const args = [
    // Force DASH video + audio, fallback to progressive MP4
    "-f", "bv*[ext=mp4]+ba[ext=m4a]/b[ext=mp4]",

    // Merge to single MP4 file
    "--merge-output-format", "mp4",

    // Avoid .part partial file confusion
    "--no-part",

    // Resilience options
    "--retries", "10",
    "--fragment-retries", "10",
    "--concurrent-fragments", "5",

    // Output location
    "-o", outputTemplate,

    // URL
    url
  ];

  console.log("[yt-dlp] Executing:", args.join(" "));

  try {
    const { stdout, stderr } = await execFileAsync("yt-dlp", args);

    console.log("[yt-dlp] stdout:", stdout);
    console.log("[yt-dlp] stderr:", stderr);

  } catch (err) {

    console.error("[yt-dlp] initial download failed");

    /**
     * If 403 or login-related errors appear,
     * retry using cookies from Firefox.
     *
     * We do NOT assume cookies are needed.
     * We only fallback if first attempt fails.
     */

    if (err.stderr && err.stderr.includes("403")) {

      console.log("[yt-dlp] Retrying with browser cookies...");

      const retryArgs = [
        "--cookies-from-browser", "firefox",
        ...args
      ];

      const { stdout, stderr } = await execFileAsync("yt-dlp", retryArgs);

      console.log("[yt-dlp] retry stdout:", stdout);
      console.log("[yt-dlp] retry stderr:", stderr);

    } else {
      throw err;
    }
  }

  /**
   * After download completes, we verify:
   *  - File exists
   *  - File size > 1MB
   */

  const files = fs.readdirSync(tempDir)
    .filter(f => f.startsWith("youtube-") && f.endsWith(".mp4"))
    .map(f => ({
      file: f,
      full: path.join(tempDir, f),
      size: fs.statSync(path.join(tempDir, f)).size,
      mtime: fs.statSync(path.join(tempDir, f)).mtime.getTime()
    }))
    .sort((a, b) => b.mtime - a.mtime);

  if (!files.length) {
    throw new Error("Download completed but no MP4 found");
  }

  const latest = files[0];

  if (latest.size < 1024 * 1024) {
    throw new Error("Downloaded file suspiciously small");
  }

  console.log("[yt-dlp] Verified file:", latest.full);
  console.log("[yt-dlp] Size:", latest.size);

  return latest.full;
}

async function launchPlayer(playerCommand, filePath) {
