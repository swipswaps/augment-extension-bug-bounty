const fs = require("fs");
const { spawnSync } = require("child_process");
const path = require("path");

// CONFIGURATION
const MAX_RETRIES = 5; // RULE 2, 15
const RETRY_DELAY_MS = 2000; // RULE 2
const LOG_DIR = ".notes"; // RULE 22
const WATCHDOG_SCRIPT = ".augment/scripts/terminal-watchdog.sh"; // RULE 9
const PRE_RESPONSE_CHECK = ".augment/scripts/pre-response-check.sh"; // RULE 9

// UTILITY FUNCTIONS
function fail(msg) {
  console.error(`❌ FATAL: ${msg}`); // RULE 7
  process.exit(1);
}

function logVerbose(message) {
  console.log(`[VERBOSE] ${message}`); // RULE 12, 22
}

function runCommand(command, options = {}) {
  const logfile =
    options.logfile ||
    path.join(LOG_DIR, `terminal-${new Date().toISOString().replace(/[:.]/g, "-")}.log`);

  const fullCommand = `${command} 2>&1 | tee -a "${logfile}"`; // RULE 22 mandatory terminal logging
  logVerbose(`Running: ${fullCommand}`); // RULE 8

  const result = spawnSync("bash", ["-c", fullCommand], { encoding: "utf8" });

  // RULE 9: Terminal watchdog enforcement
  spawnSync("bash", [WATCHDOG_SCRIPT], { stdio: "inherit" });

  if (result.error) fail(`Execution failed during runCommand: ${result.error.message}`); // RULE 7, 15

  const output = result.stdout || "";
  const error = result.stderr || "";
  logVerbose(`Command output captured: ${output.trim().slice(0, 120)}...`); // RULE 7, 8
  return { output, error, logfile }; // RULE 8
}

// EXTRACT <output> BLOCKS FROM INPUT
function extractBlocks(input) {
  const regex = /<output>([\s\S]*?)<\/output>/g;
  const blocks = [];
  let match;
  while ((match = regex.exec(input)) !== null) {
    blocks.push(match[1].trim());
  }
  if (blocks.length !== 2) fail(`Expected exactly 2 <output> blocks, found: ${blocks.length}`); // RULE 2
  return blocks;
}

// VALIDATE HEADERS
function validateHeaders(blocks) {
  blocks.forEach((block, idx) => {
    if (!block.startsWith("//")) fail(`Block ${idx + 1} missing mandatory comment header`); // RULE 6
  });
}

// WRITE FILES TO DISK
function writeFiles(blocks) {
  const files = [
    { name: "instrument-closing-promise.js", content: blocks[0] },
    { name: "launch-instrumented-augment.sh", content: blocks[1] },
  ];
  files.forEach((file) => {
    fs.writeFileSync(file.name, file.content, { encoding: "utf8" }); // RULE 1, 11
    if (file.name.endsWith(".sh")) fs.chmodSync(file.name, 0o755); // RULE 20
    logVerbose(`File written: ${file.name}`); // RULE 7, 8
  });
}

// VERIFY LOG FOR SUCCESS
function verifyLog(logfile) {
  if (!fs.existsSync(logfile)) fail(`Log file missing: ${logfile}`); // RULE 22
  const content = fs.readFileSync(logfile, "utf8");
  if (!content.includes("_closingPromise") || !content.includes("STACK TRACE")) return false; // RULE 7
  return true;
}

// EXECUTION WITH RETRIES
function executeWithRetries() {
  let attempt = 0;
  while (attempt < MAX_RETRIES) {
    attempt++;
    logVerbose(`Attempt ${attempt} of ${MAX_RETRIES}`); // RULE 12, 15
    const { output, logfile } = runCommand("./launch-instrumented-augment.sh"); // RULE 6, 9

    // RULE 9: Pre-response check
    spawnSync("bash", [PRE_RESPONSE_CHECK], { stdio: "inherit" });

    // RULE 7: Verify execution via log
    if (verifyLog(logfile)) {
      console.log("✅ Execution verified successfully."); // RULE 7
      return;
    } else {
      console.warn(
        `⚠️ Verification failed on attempt ${attempt}. Retrying in ${RETRY_DELAY_MS}ms...`
      ); // RULE 7, 15
      Atomics.wait(new Int32Array(new SharedArrayBuffer(4)), 0, 0, RETRY_DELAY_MS); // RULE 2
    }
  }
  fail(`All ${MAX_RETRIES} attempts failed. Check terminal logs for details.`); // RULE 15
}

// MAIN EXECUTION
if (require.main === module) {
  console.log("RETRY-ENFORCING COMPLIANCE DRIVER");
  console.log("=================================\n");

  const input = fs.readFileSync(0, "utf8"); // RULE 9

  if (!input || input.trim() === "") {
    fail(
      "Empty <output> block detected at transport layer. RULE 9 violation: Mandatory output reading failed."
    ); // RULE 9
  }

  const blocks = extractBlocks(input); // RULE 9
  validateHeaders(blocks); // RULE 6
  writeFiles(blocks); // RULE 1
  executeWithRetries(); // RULE 2, 15

  console.log("✅ All steps completed successfully. PRF & RULES 0-22 enforced."); // RULE 0, 1, 2
}
