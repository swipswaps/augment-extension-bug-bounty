const { spawn } = require("child_process");

function streamVideoDirectly(url) {
  console.log("[STREAM] Starting true streaming pipeline");

  // Spawn yt-dlp with stdout streaming
  const ytdlp = spawn("yt-dlp", [
    "-f", "bestvideo+bestaudio/best",
    "-o", "-",               // IMPORTANT: send output to stdout
    "--no-part",             // don't create .part file
    url
  ], {
    stdio: ["ignore", "pipe", "pipe"]  // pipe stdout
  });

  // Spawn mpv reading from stdin
  const mpv = spawn("mpv", [
    "--cache=yes",                 // enable internal cache
    "--demuxer-max-bytes=100M",    // allow large demux buffer
    "--demuxer-readahead-secs=30", // read ahead 30 seconds
    "--cache-secs=20",             // 20 sec playback cache
    "--force-seekable=yes",        // treat stream as seekable
    "--profile=low-latency",       // reduce startup delay
    "--no-terminal",               // cleaner logs
    "-"                            // read from stdin
  ], {
    stdio: ["pipe", "ignore", "ignore"]
  });

  // Pipe yt-dlp stdout → mpv stdin
  ytdlp.stdout.pipe(mpv.stdin);

  // Log errors clearly
  ytdlp.stderr.on("data", data => {
    console.log("[yt-dlp]", data.toString());
  });

  mpv.on("close", code => {
    console.log("[mpv] exited with code", code);
  });

  ytdlp.on("close", code => {
    console.log("[yt-dlp] exited with code", code);
  });
}
