/**
 * PROGRESSIVE DOWNLOAD WITH EARLY PLAYBACK
 *
 * This implementation:
 * - Starts yt-dlp in background
 * - Writes directly to final MP4 file
 * - Monitors file size
 * - Launches mpv once threshold reached
 * - Continues download in background
 *
 * Inspired by:
 * - mpv growing-file support
 * - yt-dlp progressive MP4 download behavior
 */

const { spawn } = require("child_process");

async function progressiveYouTubePlayback(url, playerCommand) {

  const tempDir = require("os").tmpdir();
  const outputFile = require("path").join(
    tempDir,
    `youtube-${Date.now()}.mp4`
  );

  const args = [
    "-f", "18/bv*[ext=mp4][protocol!=m3u8]+ba[ext=m4a][protocol!=m3u8]/b[ext=mp4][protocol!=m3u8]",
    "--merge-output-format", "mp4",
    "--no-part",
    "-o", outputFile,
    url
  ];

  console.log("[yt-dlp] Starting progressive download");

  const ytdlp = spawn("yt-dlp", args);

  ytdlp.stderr.on("data", d =>
    console.log("[yt-dlp]", d.toString())
  );

  const MIN_PLAYABLE_SIZE = 5 * 1024 * 1024; // 5MB
  let playerStarted = false;

  /**
   * Poll file size every 1 second.
   * Once threshold reached, launch player.
   */

  const interval = setInterval(() => {
    if (!require("fs").existsSync(outputFile)) return;

    const size = require("fs").statSync(outputFile).size;

    console.log("[download-progress] bytes:", size);

    if (size > MIN_PLAYABLE_SIZE && !playerStarted) {

      console.log("[player] Launching early playback");

      const mpv = spawn(playerCommand, [
        "--force-seekable=yes",
        "--cache=yes",
        "--cache-secs=20",
        outputFile
      ], {
        detached: true,
        stdio: "ignore"
      });

      mpv.unref();

      playerStarted = true;
    }

  }, 1000);

  ytdlp.on("close", code => {
    console.log("[yt-dlp] exited with code", code);
    clearInterval(interval);
  });

  return {
    success: true,
    file: outputFile,
    progressive: true
  };
}

Replace progressive function with this:

const { spawn } = require("child_process");
const fs = require("fs");
const path = require("path");
const os = require("os");

function startResumableDownload(id, url) {

  const outputFile = path.join(os.tmpdir(), `youtube-${id}.mp4`);

  const args = [
    "-f", "18/bv*[ext=mp4][protocol!=m3u8]+ba[ext=m4a][protocol!=m3u8]/b[ext=mp4][protocol!=m3u8]",
    "--merge-output-format", "mp4",

    /**
     * KEY: Resume support
     * --continue resumes partial files
     */
    "--continue",

    /**
     * KEY: Emit structured progress lines
     */
    "--newline",

    "-o", outputFile,
    url
  ];

  const proc = spawn("yt-dlp", args);

  activeDownloads.set(id, {
    process: proc,
    progress: 0,
    filePath: outputFile,
    lastUpdate: Date.now()
  });

  proc.stdout.on("data", data => {
    const text = data.toString();

    /**
     * yt-dlp progress lines look like:
     * [download]  23.4% of 12.34MiB at 1.23MiB/s ETA 00:10
     */

    const match = text.match(/(\d+\.\d+)%/);
    if (match) {
      const percent = parseFloat(match[1]);

      const entry = activeDownloads.get(id);
      if (entry) {
        entry.progress = percent;
        entry.lastUpdate = Date.now();
      }
    }
  });

  proc.on("close", () => {
    activeDownloads.delete(id);
  });

  return outputFile;
}

  const entry = activeDownloads.get(id);

  if (!entry) {
    return res.status(404).json({ error: "Download not found" });
  }

  entry.process.kill("SIGTERM");

  activeDownloads.delete(id);

  return res.json({ success: true, cancelled: id });
});

/**
 * STALL DETECTION
 *
 * If no progress update for 30 seconds,
 * log and optionally restart download.
 */

setInterval(() => {

  const now = Date.now();

  for (const [id, entry] of activeDownloads.entries()) {

    if (now - entry.lastUpdate > 30000) {

      console.warn("[STALL DETECTED]", id);

      entry.process.kill("SIGTERM");

      activeDownloads.delete(id);
    }
  }

}, 10000);
