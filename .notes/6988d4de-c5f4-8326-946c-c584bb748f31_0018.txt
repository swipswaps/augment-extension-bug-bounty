run-tool.sh (Deterministic State Machine Version)
#!/usr/bin/env bash
set -euo pipefail

CMD="$1"
RUN_ID="$(date +%s%N)"
BASE="$HOME/.edc/runs/$RUN_ID"

mkdir -p "$BASE"

STATE_FILE="$BASE/state.txt"
META_FILE="$BASE/meta.txt"
STDOUT_FILE="$BASE/stdout.txt"
STDERR_FILE="$BASE/stderr.txt"

echo "NEW" > "$STATE_FILE"

log_state() {
  echo "$1" > "$STATE_FILE"
}

log_meta() {
  echo "$1" >> "$META_FILE"
}

log_meta "RUN_ID=$RUN_ID"
log_meta "COMMAND=$CMD"
log_meta "START_TIME=$(date +%s)"

log_state "QUEUED"

(
  log_state "RUNNING"
  bash -c "$CMD"
) >"$STDOUT_FILE" 2>"$STDERR_FILE" &

PID=$!

TIMEOUT=15
GRACE=3

(
  sleep "$TIMEOUT"
  if kill -0 "$PID" 2>/dev/null; then
    log_state "TIMEOUT"
    kill -TERM "$PID"
    sleep "$GRACE"
    kill -KILL "$PID" 2>/dev/null || true
  fi
) &

WATCHDOG_PID=$!

wait "$PID" 2>/dev/null || true
EXIT_CODE=$?

kill "$WATCHDOG_PID" 2>/dev/null || true

if grep -q "TIMEOUT" "$STATE_FILE"; then
  log_meta "EXIT_CODE=TIMEOUT"
else
  log_state "COMPLETED"
  log_meta "EXIT_CODE=$EXIT_CODE"
fi

log_meta "END_TIME=$(date +%s)"


This guarantees:

stdout file closed before state finalization

deterministic ordering

timeout recorded before kill

no Promise race

no in-memory return dependency
