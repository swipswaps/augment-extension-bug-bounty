3️⃣ PARSE RESULTS (NODE.JS WRAPPER)

We parse the file outputs and report a structured result.

📄 parse-result.js

import * as fs from "fs"
import * as path from "path"

export function parseRun(runMetaPath) {
  const content = fs.readFileSync(runMetaPath, "utf8")
  const lines = content.trim().split("\n")
  const meta = Object.fromEntries(lines.map(l => l.split("=")))
  return meta
}

export function loadOutput(runId) {
  const base = path.join(process.env.HOME, ".edc/runs", runId)
  const meta = parseRun(path.join(base, "meta.txt"))
  const stdout = fs.readFileSync(path.join(base, "stdout.txt"), "utf8")
  const stderr = fs.readFileSync(path.join(base, "stderr.txt"), "utf8")
  return {
    state: meta.state,
    exitCode: Number(meta.exit_code),
    stdout,
    stderr,
    runtime: Number(meta.end) - Number(meta.start),
  }
}


This isolates all output reliably, avoiding any timing races.
